**2. Create `index.tsx`** -- WXT content script entry point with createShadowRootUi:

```typescript
import { createShadowRootUi } from 'wxt/client'
import ReactDOM from 'react-dom/client'
import PreviewPanel from './App'

export default defineContentScript({
  matches: [
    '*://boards.greenhouse.io/*/jobs/*',
    '*://boards.greenhouse.io/*/apply/*',
    '*://job-boards.greenhouse.io/*/jobs/*',
  ],
  cssInjectionMode: 'ui',

  async main(ctx) {
    let uiInstance: ReturnType<typeof createShadowRootUi> extends Promise<infer T> ? T : never

    // Listen for FILL_STARTED message from background worker
    chrome.runtime.onMessage.addListener(async (msg) => {
      if (msg.type === 'FILL_STARTED' && msg.payload?.profileId) {
        const profileId = msg.payload.profileId

        // Request data from background worker
        const [profileData, mappingConfig, userIdentity, duplicateCheck] = await Promise.all([
          chrome.runtime.sendMessage({ action: 'getProfileForFill', payload: { profileId } }),
          chrome.runtime.sendMessage({ action: 'getFieldMappings', payload: { platform: 'greenhouse' } }),
          new Promise(resolve => chrome.storage.local.get(['userIdentity'], r => resolve(r.userIdentity))),
          chrome.runtime.sendMessage({ action: 'checkDuplicateApplication', payload: { applyUrl: window.location.href } }),
        ])

        // Scan form fields
        const { scanGreenhouseForm } = await import('../../lib/greenhouse/scanner')
        const fields = scanGreenhouseForm()

        // Map profile to fields
        const { mapProfileToFields } = await import('../../lib/greenhouse/mapper')
        const mappedFields = mapProfileToFields(
          fields,
          profileData?.applicationProfile || {},
          userIdentity || { email: '', fullName: null, phone: null, portfolioUrl: null, location: null },
          mappingConfig || { platform: 'greenhouse', version: 1, mappings: [], is_active: true }
        )

        // Mount shadow DOM UI
        uiInstance = await createShadowRootUi(ctx, {
          name: 'autoapply-preview',
          position: 'overlay',
          anchor: 'body',
          onMount: (container) => {
            const app = document.createElement('div')
            container.append(app)
            const root = ReactDOM.createRoot(app)
            root.render(
              <PreviewPanel
                mappedFields={mappedFields}
                profileName={profileData?.name || 'Profile'}
                profileId={profileId}
                duplicateInfo={duplicateCheck}
                onDismiss={() => uiInstance?.remove()}
              />
            )
            return root
          },
          onRemove: (root) => root?.unmount(),
        })
        uiInstance.mount()
      }
    })
  },
})
```
